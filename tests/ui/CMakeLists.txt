# UI tests: the plugin's Qt dialogs, run without OBS. They need Qt 6 and the
# libobs headers (only obs_module_text / obs_module_file are used, and
# obs_stubs.cpp provides them). On macOS both come from the .deps folder the
# plugin's first configure downloads:
#
#   cmake -S tests/ui -B build_ui_tests -G Ninja
#   cmake --build build_ui_tests
#   ctest --test-dir build_ui_tests --output-on-failure
#
# Elsewhere, point LANGLIVE_QT_PREFIX and LANGLIVE_OBS_SOURCE_DIR at a Qt 6
# install and an obs-studio source tree.

cmake_minimum_required(VERSION 3.28)

project(langlive-obs-plugin-ui-tests LANGUAGES C CXX)

set(_root "${CMAKE_CURRENT_SOURCE_DIR}/../..")

# The same Qt and obs-studio versions the plugin builds against.
file(READ "${_root}/buildspec.json" _buildspec)
string(JSON _qt_version GET "${_buildspec}" dependencies qt6 version)
string(JSON _obs_version GET "${_buildspec}" dependencies obs-studio version)
file(GLOB _default_qt_prefix "${_root}/.deps/obs-deps-qt6-${_qt_version}-*")
set(_default_obs_source "${_root}/.deps/obs-studio-${_obs_version}")

set(LANGLIVE_QT_PREFIX "${_default_qt_prefix}" CACHE PATH "Qt 6 install prefix")
set(LANGLIVE_OBS_SOURCE_DIR "${_default_obs_source}" CACHE PATH "obs-studio source tree (for libobs headers)")

list(PREPEND CMAKE_PREFIX_PATH "${LANGLIVE_QT_PREFIX}")
find_package(Qt6 REQUIRED COMPONENTS Widgets)

# obsconfig.h is generated by the obs-studio build the plugin's configure runs.
file(GLOB_RECURSE _obsconfig "${LANGLIVE_OBS_SOURCE_DIR}/build_*/config/obsconfig.h")
list(GET _obsconfig 0 _obsconfig)
get_filename_component(_obsconfig_dir "${_obsconfig}" DIRECTORY)

enable_testing()

# Also builds the core tests and provides doctest::doctest.
set(LANGLIVE_CORE_BUILD_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${_root}/core" core)

set(_src "${_root}/src")
add_executable(langlive-ui-tests)
set_target_properties(langlive-ui-tests PROPERTIES AUTOMOC ON AUTORCC ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_sources(
  langlive-ui-tests
  PRIVATE
    main.cpp
    obs_stubs.cpp
    ui_test_support.hpp
    dialogs_test.cpp
    go_live_dialog_test.cpp
    screenshots.cpp
    langlive_api_json_test.cpp
    qa_override_test.cpp
    start_failure_watcher_test.cpp
    secure_store_test.cpp
    ${_src}/obs/qa-override.cpp
    ${_src}/obs/start-failure-watcher.cpp
    ${_src}/app/langlive-api.cpp
    ${_src}/app/mock-langlive-api.cpp
    ${_src}/app/plugin-controller.cpp
    ${_src}/obs/langlive-api-json.cpp
    ${_src}/ui/cover-picker.cpp
    ${_src}/ui/cover-picker.hpp
    ${_src}/ui/diagnostics-dialog.cpp
    ${_src}/ui/diagnostics-dialog.hpp
    ${_src}/ui/go-live-dialog.cpp
    ${_src}/ui/go-live-dialog.hpp
    ${_src}/ui/langlive-menu.cpp
    ${_src}/ui/langlive-menu.hpp
    ${_src}/ui/login-dialog.cpp
    ${_src}/ui/login-dialog.hpp
    ${_src}/ui/message-dialog.cpp
    ${_src}/ui/preflight-dialog.cpp
    ${_src}/ui/preflight-dialog.hpp
    ${_src}/ui/qr-code-widget.cpp
    ${_src}/ui/qr-code-widget.hpp
    ${_src}/ui/resources/ui.qrc
    ${_src}/ui/style.cpp
    ${_src}/ui/text.cpp
    ${_src}/ui/update-dialog.cpp
    ${_src}/ui/update-dialog.hpp
    ${_root}/third_party/qrcodegen/qrcodegen.cpp
)
target_include_directories(
  langlive-ui-tests
  PRIVATE "${_src}" "${_root}/tests" "${_root}/third_party/qrcodegen" "${LANGLIVE_OBS_SOURCE_DIR}/libobs" "${_obsconfig_dir}"
)
target_compile_definitions(langlive-ui-tests PRIVATE LANGLIVE_TEST_DATA_DIR="${_root}/data")
target_link_libraries(langlive-ui-tests PRIVATE Qt6::Widgets langlive::obsplugin::core doctest::doctest)

if(APPLE)
  include("${_root}/cmake/macos/agl-stub.cmake")
  langlive_link_agl_stub(langlive-ui-tests)
  target_sources(langlive-ui-tests PRIVATE ${_src}/obs/secure-store-macos.cpp)
  target_link_libraries(langlive-ui-tests PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
  target_sources(langlive-ui-tests PRIVATE ${_src}/obs/secure-store-windows.cpp)
  target_link_libraries(langlive-ui-tests PRIVATE Advapi32)
endif()
if(WIN32)
  # The libobs headers declare its functions dllimport; obs_stubs.cpp defines
  # them in this executable instead.
  target_compile_definitions(langlive-ui-tests PRIVATE IS_LIBOBS)
endif()

# No display needed: Qt's offscreen platform (OBS's Windows Qt has only the
# "minimal" headless platform; main.cpp picks it).
add_test(NAME langlive-ui-tests COMMAND langlive-ui-tests)
set_tests_properties(langlive-ui-tests PROPERTIES ENVIRONMENT "QT_PLUGIN_PATH=${LANGLIVE_QT_PREFIX}/plugins")
if(WIN32)
  set_tests_properties(langlive-ui-tests PROPERTIES ENVIRONMENT_MODIFICATION
                                                    "PATH=path_list_prepend:${LANGLIVE_QT_PREFIX}/bin")
endif()
